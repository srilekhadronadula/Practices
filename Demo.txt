    public static void main(String[] args) {

        System.out.println("enter the n value");
        int n=sc.nextInt();
        int i,j,b;
        int arr[]=new int[n];
        for(i=0;i<n;i++){
            arr[i]=sc.nextInt();
        }
       int l[]=new int[n];
       int r[]=new int[n];
       l[0]=arr[0];
       for(i=1;i<n;i++){
          l[i]=Math.max(l[i],l[i-1]); 
         
       }
       for(i=n-1;i>=0;i--){
           r[i]=Math.max(r[n-1],r[n-2]);
       }
      for(i=0;i<n;i++){
      b=Math.max(l[i],r[i])-arr[i];
      
      System.out.println(b);}
    }
}

}

        System.out.println("dog sleeping");
    }
}
