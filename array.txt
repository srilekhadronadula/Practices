#sorting
#include<stdio.h>
int main(){
    int a[40],i,j,n,temp;
    printf("enter the array length");
    scanf(" %d",&n);
    for(i=0;i<n;i++){
        printf("enter the elements");
        scanf("%d",&a[i]);
    }
    for(i=0;i<n;i++){
        for(j=0;j<n;j++){
            if(a[i]<a[j]){
                temp=a[i];
                a[i]=a[j];
                a[j]=temp;
            }
        }
    }
    printf("sorted array");
    for(j=0;j<n;j++){
        printf("%d",a[j]);
    }
}

#long common string
#include <stdio.h>
#include<string.h>
int main() {
   char a[50][50],result[50];
   int i,j,len,n;
   printf("enter the length of the array");
   scanf("%d",&n);
   for(i=0;i<n;i++){
       printf("enter the string:");
       scanf("%s",&a[i]);
   }
   len=strlen(a[0]);
   for(i=0;i<n;i++){
       for(j=1;a[i]!='\0';j++){
           if(a[j][i]!=a[0][i]){
           len=i;
           break;
       }
       }
   }
   for(i=0;i<len;i++){
       result[i]=a[0][i];
   }
   printf("%s",result);
}
